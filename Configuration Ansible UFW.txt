- name: Configuration complémentaire du serveur web
  hosts: all
  become: yes

  tasks:
    - name: Installer UFW
      apt:
        name: ufw
        state: present
        update_cache: yes

    - name: Autoriser SSH
      ufw:
        rule: allow
        port: 22

    - name: Autoriser HTTP
      ufw:
        rule: allow
        port: 80

    - name: Activer le pare-feu UFW
      ufw:
        state: enabled

    - name: Créer une page d’accueil HTML
      copy:
        dest: /var/www/html/index.html
        content: |
          <html>
          <head><title>Bienvenue</title></head>
          <body><h1>Serveur Web NGINX opérationnel</h1></body>
          </html>

    - name: Redémarrer NGINX pour recharger les fichiers
      service:
        name: nginx
        state: restarted
